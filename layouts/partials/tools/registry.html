{{- /* The machine-readable tool registry, built from the same data the site renders. Returns a slice of tool dicts, newest catalog order preserved. Sources, in order of precedence: data/tools.yaml id, name, description, keywords, category, status data/tool-guides.yaml about + FAQ (the FAQ questions become examples) data/tool-ai.yaml optional use_cases / examples / keywords / features */ -}} {{- $guides := index site.Data "tool-guides" | default dict -}} {{- $overlay := index site.Data "tool-ai" | default dict -}} {{- $base := site.BaseURL -}} {{- $defaultFeatures := slice "Runs in the browser" "No login required" -}} {{- $thirdPartyTools := slice "cors-checker" "dns-lookup" "grpc-tester" "http-request-tester" "ip-lookup" "webhook-tester" "whois-lookup" -}} {{- $tools := slice -}} {{- range site.Data.tools -}} {{- $category := . -}} {{- range .tools -}} {{- $slug := .slug -}} {{- $guide := index $guides $slug | default dict -}} {{- $ai := index $overlay $slug | default dict -}} {{- $examples := slice -}} {{- range ($guide.faq | default slice) -}} {{- $examples = $examples | append .q -}} {{- end -}} {{- with index $ai "examples" }}{{ $examples = . }}{{ end -}} {{- /* Keywords are free text in YAML, so a bare 2048 or 443 parses as a number. Coerce to strings here so tools.json is type-stable. */ -}} {{- $keywords := slice -}} {{- range (.keywords | default slice) -}} {{- $keywords = $keywords | append (printf "%v" .) -}} {{- end -}} {{- $canonicalURL := printf "%stools/%s/" $base $slug -}} {{- $markdownURL := printf "%stools/%s/index.md" $base $slug -}} {{- $features := index $ai "features" | default $defaultFeatures -}} {{- $processing := "client-side" -}} {{- $dataUploaded := false -}} {{- $requiresBackend := false -}} {{- if in $thirdPartyTools $slug -}} {{- $processing = "third-party-api" -}} {{- $dataUploaded = true -}} {{- end -}} {{- with index $ai "privacy" -}} {{- with .processing }}{{ $processing = . }}{{ end -}} {{- if isset . "data_uploaded" }}{{ $dataUploaded = .data_uploaded }}{{ end -}} {{- if isset . "requires_backend" }}{{ $requiresBackend = .requires_backend }}{{ end -}} {{- end -}} {{- $offlineCapable := not $dataUploaded -}} {{- if isset $ai "offline_capable" }}{{ $offlineCapable = index $ai "offline_capable" }}{{ end -}} {{- $privacy := dict "processing" $processing "data_uploaded" $dataUploaded -}} {{- with index $ai "privacy" }}{{ with .provider }}{{ $privacy = merge $privacy (dict "provider" .) }}{{ end }}{{ end -}} {{- $entry := dict "id" $slug "name" (printf "%v" .name) "url" $canonicalURL "canonical_url" $canonicalURL "documentation" $markdownURL "markdown_url" $markdownURL "api_url" nil "mcp_available" true "client_side" (not $requiresBackend) "offline_capable" $offlineCapable "requires_backend" $requiresBackend "requires_login" false "requires_api_key" false "stores_user_data" false "category" $category.slug "category_name" $category.name "description" (printf "%v" .desc) "keywords" $keywords "examples" $examples "features" $features "use_cases" (index $ai "use_cases" | default slice) "input" (index $ai "input" | default (dict "type" "text")) "output" (index $ai "output" | default (dict "type" "text")) "privacy" $privacy "limitations" (index $ai "limitations" | default slice) "related_tools" (index $ai "related_tools" | default slice) "status" (.status | default "done") -}} {{- with index $ai "keywords" }}{{ $entry = merge $entry (dict "keywords" .) }}{{ end -}} {{- $tools = $tools | append $entry -}} {{- end -}} {{- end -}} {{- return $tools -}}
